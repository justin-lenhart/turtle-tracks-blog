{{/* Outputs every image in assets/img/favorites/ as a gallery tile, resized for the web. There is no list to edit. Add a file to the folder and it appears. Delete a file and it disappears. The shortcode sorts tiles by filename, so a "01-" prefix puts a photo first. Use this shortcode INSIDE the theme's gallery block. The gallery block supplies the grid styling. In content/_index.md: {{< gallery >}}{{< favorites >}}{{< /gallery >}} {{< gallery >}}{{< favorites limit="9" >}}{{< /gallery >}} first 9 only Tile width: grid-w50 = 2 per row on phones. md:grid-w33 = 3 per row on larger screens. Use md:grid-w25 for 4 per row. */}} {{ $limit := int (.Get "limit" | default 0) }} {{ $images := sort (resources.Match "img/favorites/*.{jpg,jpeg,png,webp}") "Name" }} {{ if gt $limit 0 }}{{ $images = first $limit $images }}{{ end }} {{/* Do not add loading="lazy". The theme's layout script measures tile heights on window load. Lazy images are not loaded at that time, so each tile measures 0 px and all tiles stack at the top. The width and height attributes let the browser reserve the correct space before the file loads. */}} {{ range $images }} {{ $small := .Resize "800x q78" }} {{ end }}
